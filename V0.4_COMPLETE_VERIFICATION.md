
### Run CLI
```bash
cd C:\melodie\apps\melodie-cli
melodie init        # Initialize system
melodie verify      # Verify integrity
melodie status      # View metrics
melodie revenue     # Check revenue
```

### Access API
```bash
# Health check
curl http://localhost:3000/health

# Record game event (server-authoritative)
curl -X POST http://localhost:3000/api/v1/game/event \
  -H "Content-Type: application/json" \
  -H "x-event-id: $(uuidgen)" \
  -d '{"userId":"user1","eventType":"BOSS_KILL","clientValue":999999}'
# Response: {"karmaAwarded": 100} (server calculated, client value ignored)

# View ledger
curl http://localhost:3000/api/v1/game/ledger?userId=user1

# View reward table
curl http://localhost:3000/api/v1/game/info
```
